const express = require("express"); const cors = require("cors"); require("dotenv").config(); const assessmentRoutes = require("./routes/assessment"); const app = express(); const PORT = process.env.PORT || 5000; app.use(cors()); app.use(express.json()); // Home route app.get("/", (req, res) => { res.json({ message: "MediKiosk Backend is running successfully!" }); }); // Health check app.get("/api/health", (req, res) => { res.json({ status: "healthy", project: "MediKiosk" }); }); // Assessment API app.use("/api/assessment", assessmentRoutes); app.listen(PORT, () => { console.log(`MediKiosk server running on port ${PORT}`); });
